# The adapter is compiled alone: no game/base object dependency or game runtime.
if(NOT STAR_USE_RUST_LIGHTING)
  message(WARNING "lighting lab is comparing the C++ fallback, not Rust")
endif()
enable_testing()
find_package(Threads REQUIRED)
# Link the single aggregate explicitly only for Rust adapter tests. Keeping it
# out of the shared list also makes the C++ fallback a Rust-free link boundary.
set(lab_ext_libs ${STAR_EXT_LIBS})
list(REMOVE_ITEM lab_ext_libs star_rust_lighting star_rust_engine_ffi)
foreach(lab_target lighting_lab lighting_lab_fastmath lighting_lab_cpp)
  add_executable(${lab_target}
    LightingLab.cpp
    oracle/StarCellularLightArray.cpp
    ${PROJECT_SOURCE_DIR}/base/StarCellularLightArray.cpp
    $<TARGET_OBJECTS:star_core>)
  target_include_directories(${lab_target} PRIVATE ${STAR_CORE_INCLUDES} ${STAR_BASE_INCLUDES})
  target_link_libraries(${lab_target} PRIVATE star_extern ${lab_ext_libs} Threads::Threads)
  if(lab_target STREQUAL "lighting_lab_cpp")
    # Directory-wide definitions must not silently turn this into another Rust test.
    if(MSVC)
      target_compile_options(${lab_target} PRIVATE /USTAR_USE_RUST_LIGHTING)
    else()
      target_compile_options(${lab_target} PRIVATE -USTAR_USE_RUST_LIGHTING)
    endif()
    target_compile_definitions(${lab_target} PRIVATE LIGHTING_LAB_EXPECT_CPP)
  elseif(STAR_USE_RUST_LIGHTING)
    target_compile_definitions(${lab_target} PRIVATE STAR_USE_RUST_LIGHTING)
    target_link_libraries(${lab_target} PRIVATE star_rust_lighting)
  endif()
  if(MSVC)
    if(NOT lab_target STREQUAL "lighting_lab_fastmath")
      target_compile_options(${lab_target} PRIVATE /fp:precise)
    else()
      target_compile_options(${lab_target} PRIVATE /fp:fast)
    endif()
  else()
    if(NOT lab_target STREQUAL "lighting_lab_fastmath")
      target_compile_options(${lab_target} PRIVATE -fno-fast-math -ffp-contract=off)
    else()
      target_compile_options(${lab_target} PRIVATE -ffast-math)
    endif()
  endif()
  target_compile_definitions(${lab_target} PRIVATE LIGHTING_LAB_MODE="${lab_target}")
  add_test(NAME ${lab_target}_differential COMMAND ${lab_target} --seed 1592594996 --random 128)
  add_test(NAME ${lab_target}_self_test COMMAND ${lab_target} --self-test)
  add_test(NAME ${lab_target}_known_failure COMMAND ${lab_target} --inject-mismatch --random 0)
  set_tests_properties(${lab_target}_known_failure PROPERTIES WILL_FAIL TRUE)
endforeach()
